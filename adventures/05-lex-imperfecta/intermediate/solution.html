Key Patch: Aegyptus-Only Rule

aegyptus-require-scribe-role.yaml

Key Patch: Exception Guard

aegyptus-legacy-workload.yaml

matchConditions: - name: in-aegyptus expression: object.metadata.namespace == 'aegyptus' - name: is-legacy expression: | has(object.metadata.labels) && 'republic.rome/legacy' in object.metadata.labels && object.metadata.labels['republic.rome/legacy'] == 'true'
